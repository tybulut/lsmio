include(GNUInstallDirs)

# Define installed canonical target locations
set(LSMIOTOOL_INSTALLED_PYTHON_DIR "${CMAKE_INSTALL_FULL_DATADIR}/lsmio/python")
set(LSMIOTOOL_INSTALLED_WORKER_DIR "${CMAKE_INSTALL_FULL_LIBEXECDIR}/lsmio")
set(LSMIOTOOL_INSTALLED_ETC_DIR "${CMAKE_INSTALL_FULL_DATADIR}/lsmio/etc")
set(LSMIOTOOL_INSTALLED_ASSETS_DIR "${CMAKE_INSTALL_FULL_DATADIR}/lsmio/lmp-reaxff")
set(LSMIOTOOL_INSTALLED_VERSION_FILE "${CMAKE_INSTALL_FULL_DATADIR}/lsmio/python/lsmiotool/VERSION")
set(LSMIOTOOL_INSTALLED_PROFILE_FILE "${CMAKE_INSTALL_FULL_DATADIR}/lsmio/etc/environments.json")

# Relative paths from CMAKE_INSTALL_FULL_BINDIR (public wrapper anchor)
file(RELATIVE_PATH REL_BIN_TO_PYTHON "${CMAKE_INSTALL_FULL_BINDIR}" "${LSMIOTOOL_INSTALLED_PYTHON_DIR}")
file(RELATIVE_PATH REL_BIN_TO_LIBEXEC_WORKER "${CMAKE_INSTALL_FULL_BINDIR}" "${LSMIOTOOL_INSTALLED_WORKER_DIR}/lsmiotool-worker")
file(RELATIVE_PATH REL_BIN_TO_PROFILE "${CMAKE_INSTALL_FULL_BINDIR}" "${LSMIOTOOL_INSTALLED_PROFILE_FILE}")
file(RELATIVE_PATH REL_BIN_TO_ETC "${CMAKE_INSTALL_FULL_BINDIR}" "${LSMIOTOOL_INSTALLED_PROFILE_FILE}")
file(RELATIVE_PATH REL_BIN_TO_ASSETS "${CMAKE_INSTALL_FULL_BINDIR}" "${LSMIOTOOL_INSTALLED_ASSETS_DIR}")
file(RELATIVE_PATH REL_BIN_TO_VERSION "${CMAKE_INSTALL_FULL_BINDIR}" "${LSMIOTOOL_INSTALLED_VERSION_FILE}")

# Relative paths from LSMIOTOOL_INSTALLED_WORKER_DIR (private worker wrapper anchor)
file(RELATIVE_PATH REL_LIBEXEC_TO_PYTHON "${LSMIOTOOL_INSTALLED_WORKER_DIR}" "${LSMIOTOOL_INSTALLED_PYTHON_DIR}")
file(RELATIVE_PATH REL_LIBEXEC_TO_WORKER "${LSMIOTOOL_INSTALLED_WORKER_DIR}" "${LSMIOTOOL_INSTALLED_WORKER_DIR}/lsmiotool-worker")
file(RELATIVE_PATH REL_LIBEXEC_TO_PROFILE "${LSMIOTOOL_INSTALLED_WORKER_DIR}" "${LSMIOTOOL_INSTALLED_PROFILE_FILE}")
file(RELATIVE_PATH REL_LIBEXEC_TO_ETC "${LSMIOTOOL_INSTALLED_WORKER_DIR}" "${LSMIOTOOL_INSTALLED_PROFILE_FILE}")
file(RELATIVE_PATH REL_LIBEXEC_TO_ASSETS "${LSMIOTOOL_INSTALLED_WORKER_DIR}" "${LSMIOTOOL_INSTALLED_ASSETS_DIR}")
file(RELATIVE_PATH REL_LIBEXEC_TO_VERSION "${LSMIOTOOL_INSTALLED_WORKER_DIR}" "${LSMIOTOOL_INSTALLED_VERSION_FILE}")

# Configure public wrapper
configure_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/lsmiotool-installed.in"
  "${CMAKE_CURRENT_BINARY_DIR}/lsmiotool-installed"
  @ONLY
)

# Configure private worker wrapper
configure_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/lsmiotool-worker-installed.in"
  "${CMAKE_CURRENT_BINARY_DIR}/lsmiotool-worker-installed"
  @ONLY
)

# Install public wrapper to bindir
install(
  PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/lsmiotool-installed"
  DESTINATION "${CMAKE_INSTALL_BINDIR}"
  RENAME "lsmiotool"
  PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE
    GROUP_READ GROUP_EXECUTE
    WORLD_READ WORLD_EXECUTE
)

# Install private worker wrapper to libexecdir/lsmio
install(
  PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/lsmiotool-worker-installed"
  DESTINATION "${CMAKE_INSTALL_LIBEXECDIR}/lsmio"
  RENAME "lsmiotool-worker"
  PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE
    GROUP_READ GROUP_EXECUTE
    WORLD_READ WORLD_EXECUTE
)

# Install package root __init__.py
install(
  FILES "${CMAKE_CURRENT_SOURCE_DIR}/__init__.py"
  DESTINATION "${CMAKE_INSTALL_DATADIR}/lsmio/python/lsmiotool"
  PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ
)

# Install package library modules (*.py)
file(GLOB LSMIOTOOL_LIB_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/lib/*.py")
install(
  FILES ${LSMIOTOOL_LIB_SOURCES}
  DESTINATION "${CMAKE_INSTALL_DATADIR}/lsmio/python/lsmiotool/lib"
  PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ
)

# Install VERSION file
install(
  FILES "${CMAKE_CURRENT_SOURCE_DIR}/VERSION"
  DESTINATION "${CMAKE_INSTALL_DATADIR}/lsmio/python/lsmiotool"
  PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ
)

# Install environment profiles
install(
  FILES "${CMAKE_CURRENT_SOURCE_DIR}/etc/environments.json"
  DESTINATION "${CMAKE_INSTALL_DATADIR}/lsmio/etc"
  PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ
)
install(
  FILES "${CMAKE_CURRENT_SOURCE_DIR}/etc/environments.json"
  DESTINATION "${CMAKE_INSTALL_DATADIR}/lsmio/python/lsmiotool/etc"
  PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ
)

# Install LMP benchmark assets
install(
  DIRECTORY "${CMAKE_SOURCE_DIR}/tools/bmtool/lmp-reaxff/"
  DESTINATION "${CMAKE_INSTALL_DATADIR}/lsmio/lmp-reaxff"
  FILES_MATCHING PATTERN "*"
)
install(
  FILES "${CMAKE_SOURCE_DIR}/tools/bmtool/lmp-reaxff/in.reaxc.hns"
  DESTINATION "${CMAKE_INSTALL_DATADIR}/lsmio/lmp-reaxff"
  RENAME "in.reaxff.hns"
  PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ
)
install(
  FILES "${CMAKE_SOURCE_DIR}/tools/bmtool/lmp-reaxff/data.hns-equil"
  DESTINATION "${CMAKE_INSTALL_DATADIR}/lsmio/lmp-reaxff"
  RENAME "data.hns"
  PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ
)
